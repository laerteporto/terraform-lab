##COMANDOS TECNICOS PARA O S3 - BUCKET ##

awslocal s3 ls s3://nomedobucket
# Cria um arquivo rápido de teste
echo "Teste de upload no LocalStack" > teste.txt

# Envia o arquivo para o bucket
awslocal s3 cp teste.txt s3://nomedobucket/

# Mostrar arquivo dentro do bucket
awslocal s3 ls s3://nomebucket/

#Enviar um Arquivo simulando uma Estrutura de Pastas
awslocal s3 cp teste.txt s3://bucket-localstack/uploads/documentos/teste_pasta.txt

##Baixar o Arquivo do Seu BucketPara baixar o arquivo que você acabou de subir de volta para o seu computador com outro nome
awslocal s3 cp s3://bucket-localstack/teste.txt ./arquivo_recuperado.txt

##Excluir Arquivos do Seu Bucket
awslocal s3 rm s3://bucket-localstack/teste.txt

##Limpar Todo o Bucket de uma Vez
awslocal s3 rm s3://bucket-localstack --recursive
